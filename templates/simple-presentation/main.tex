\begin{frame}[t]{Begin with the question}
  The audience should leave with one central idea and know what to do with it.

  \vspace{3mm}
  \begin{itemize}
    \item Open with the decision that matters.
    \item Let each slide advance the answer.
    \item Remove details that do not affect the conclusion.
  \end{itemize}
\end{frame}

\begin{frame}[t]{One message per slide}
  State the idea first, then support it with the evidence that matters.

  \vspace{4mm}
  \begin{block}{Example relationship}
    \[
      \text{impact} = \text{reach} \times \text{change per person}
    \]
  \end{block}
\end{frame}

\begin{frame}[t]{Make trade-offs obvious}
  \begin{table}
    \centering
    \begin{tabular}{@{}lcc@{}}
      \toprule
      & Option A & Option B \\
      \midrule
      Setup effort & Low & Medium \\
      Flexibility & Medium & High \\
      Best for & Quick start & Long-term use \\
      \bottomrule
    \end{tabular}
  \end{table}
\end{frame}

\begin{frame}[t]{End with a next step}
  \begin{enumerate}
    \item Confirm the decision or open question.
    \item Name the owner of the next action.
    \item State when the group will review progress.
  \end{enumerate}

  \vfill
  {\color{Accent}\bfseries Resolve the question raised at the start.}
\end{frame}
