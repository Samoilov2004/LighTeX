\begin{problem}[A short proof]
  Prove that the sum of two even integers is even.
\end{problem}

\begin{solution}
  Let $a=2m$ and $b=2n$ for some integers $m$ and $n$. Then
  \[
    a+b=2m+2n=2(m+n).
  \]
  Since $m+n$ is an integer, $a+b$ is even.
\end{solution}

\begin{problem}[A recurrence]
  Let $a_1=1$ and $a_{n+1}=2a_n+1$. Find a closed formula for $a_n$ and
  prove it by induction.
\end{problem}

\begin{solution}
  The first values suggest $a_n=2^n-1$. The formula holds for $n=1$.
  Assuming it holds for some $n$, we obtain
  \[
    a_{n+1}=2(2^n-1)+1=2^{n+1}-1.
  \]
  Therefore the formula holds for every positive integer $n$.
\end{solution}
